// Fonction à ajouter dans AdminModelAccess.tsx pour créer le compte admin Firebase
// À exécuter depuis la console du navigateur (F12 → Console)

async function createAdminFirebaseAccount() {
    const { auth } = await import('./firebase');
    const { createUserWithEmailAndPassword } = await import('firebase/auth');
    
    const email = 'admin@perfectmodels.online';
    const password = 'Pmm2025';
    
    try {
        const result = await createUserWithEmailAndPassword(auth, email, password);
        console.log('✅ Compte admin Firebase créé:', result.user.uid);
        alert('Compte admin créé avec succès!');
    } catch (error) {
        console.error('❌ Erreur:', error);
        alert('Erreur: ' + (error as any).message);
    }
}

// Exécuter la fonction
createAdminFirebaseAccount();
